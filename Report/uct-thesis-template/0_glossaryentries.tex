%---------------------------------------------------------------------
%   GLOSSARY TERMS: DEPTH SENSING
%---------------------------------------------------------------------
\newglossaryentry{activestereo}{
	name        = {active stereo},
	description = {Stereo depth sensing in which a projected pattern adds
		texture to the scene, while depth is still computed by
		matching between two cameras}
}
\newglossaryentry{passivestereo}{
	name        = {passive stereo},
	description = {Stereo depth sensing that relies only on the natural
		texture of the scene}
}
\newglossaryentry{structuredlight}{
	name        = {structured light},
	description = {Depth sensing that triangulates between a projector of a
		known pattern and a single camera}
}
\newglossaryentry{baseline}{
	name        = {baseline},
	description = {The distance between the optical centres of the two
		cameras of a stereo pair}
}
\newglossaryentry{disparity}{
	name        = {disparity},
	description = {The horizontal offset, in pixels, between the positions
		of the same scene point in the left and right images}
}
\newglossaryentry{depthunit}{
	name        = {depth unit},
	description = {The physical size of one step of the camera's 16-bit
		depth value, which sets the depth quantisation}
}
\newglossaryentry{visualpreset}{
	name        = {visual preset},
	description = {A predefined set of RealSense stereo-matching parameters,
		such as High Density or High Accuracy}
}
\newglossaryentry{ip65}{
	name        = {IP65},
	description = {Ingress protection rating denoting a dust-tight enclosure
		that is protected against water jets}
}

	name        = {registration},
	description = {Finding the rigid transformation that places one scan in
		the coordinate frame of another}
}
\newglossaryentry{fusion}{
	name        = {fusion},
	description = {Merging several registered scans into a single surface}
}
\newglossaryentry{voxel}{
	name        = {voxel},

}
\newglossaryentry{heightfield}{

%---------------------------------------------------------------------
%   GLOSSARY TERMS: EVALUATION AND ROUGHNESS
%---------------------------------------------------------------------
\newglossaryentry{signedbias}{
	name        = {signed bias},
	description = {The mean signed distance of reconstructed points from the
		reference surface, measuring systematic offset}
}
\newglossaryentry{precision}{
	name        = {precision},
	description = {The fraction of reconstructed points lying within a
		distance threshold of the reference surface}
}
\newglossaryentry{completeness}{
	name        = {completeness},
	description = {The fraction of points sampled from the reference surface
		that lie within a distance threshold of the
		reconstruction}
}
\newglossaryentry{fscore}{
	name        = {F-score},
	description = {The harmonic mean of precision and completeness at a given
		distance threshold}
}
\newglossaryentry{rugosity}{
	name        = {rugosity},
	description = {The ratio of the true area of a surface to its projected
		area. A flat surface has a rugosity of 1}
}
\newglossaryentry{sdr}{
	name        = {developed interfacial area ratio},
	description = {The ISO 25178-2 parameter $S_{dr}$: the percentage by
		which the true surface area exceeds the projected area}
}
\newglossaryentry{selfaffine}{
	name        = {self-affine},
	description = {Describing a surface whose roughness follows a power law
		across scales, so its power spectrum is a straight line on
		log-log axes}
}
\newglossaryentry{hurst}{
	name        = {Hurst exponent},
	description = {The exponent $H$ describing how the roughness of a
		self-affine surface changes with scale}
}
